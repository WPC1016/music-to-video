10代码使用示例（React + Remotion）
import { Emperor, Camera, FireScene, LyricBar } from './components';
import { useCurrentFrame, interpolate } from 'remotion';

export const Scene1: React.FC = () => {
  const frame = useCurrentFrame();
  // 第 45–60 帧闪光灯由 0 渐变到 1
  const cameraFlash = interpolate(frame, [45, 60], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  return (
    <div className="scene scene-1">
      <FireScene />
      <Emperor id="yongle" robe="mingHuang" expr="happy" action="shoot" />
      <Camera flash={cameraFlash} />
      <LyricBar text="房子着火我拍照" seal="永乐" />
    </div>
  );
};

├─ scenes/       # 场景 SVG（16:9）
├─ animations/   # 动画组件
└─ ui/           # 歌词条、年号印章
使用步骤
  1. 在本页悬停素材，点击「⤓ SVG」下载，放入 public/assets 对应目录；
  2. 在 React 组件中引入 SVG，或直接粘贴「复制 SVG 代码」得到的内联代码；
  3. 使用 Remotion 的 useCurrentFrame / interpolate 控制时间与动画；
  4. 根据音乐节奏调整分镜时长和动画参数（见第 9 部分时间轴）。
